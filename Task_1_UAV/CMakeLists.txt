cmake_minimum_required(VERSION 3.21)
project(Control_Work_02_Polymorphism LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets Test)
add_executable(Control_Work_02_Polymorphism WIN32 src/main.cpp src/uav.cpp src/uav.h)
target_link_libraries(Control_Work_02_Polymorphism PRIVATE Qt6::Widgets)
add_executable(model_tests tests/model_tests.cpp src/uav.cpp src/uav.h)
target_link_libraries(model_tests PRIVATE Qt6::Test)
enable_testing()
add_test(NAME model_tests COMMAND model_tests)
