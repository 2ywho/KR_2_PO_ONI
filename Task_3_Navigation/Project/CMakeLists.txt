cmake_minimum_required(VERSION 3.16)
project(Control_Work_02_Polymorphism LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTOUIC ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets Test)
add_library(model navigation.h navigation.cpp)
target_link_libraries(model PUBLIC Qt6::Core)
target_include_directories(model PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(model_tests tests.cpp)
target_link_libraries(model_tests PRIVATE model)
enable_testing()
add_test(NAME model_tests COMMAND model_tests)

qt_add_executable(NavigationLab WIN32 main.cpp)
target_link_libraries(NavigationLab PRIVATE model Qt6::Widgets)
